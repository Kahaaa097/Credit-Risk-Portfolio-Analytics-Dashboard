interface TreemapProps {
  x?: number;
  y?: number;
  width?: number;
  height?: number;
  payload?: BranchHeatmapData;
  [key: string]: unknown;
}
// Custom Treemap Content
function CustomTreemapContent(props: TreemapProps) {
  const { x = 0, y = 0, width = 0, height = 0, payload } = props;
  const { name, eop_balance, npl_ratio } = payload ?? {};

  // Màu sắc theo mức NPL
  const getColor = (ratio: number) => {
    if (ratio > 0.03) return COLORS.danger;
    if (ratio > 0.025) return COLORS.orange;
    return COLORS.green;
  };

  return (
    <g>
      <rect
        x={x}
        y={y}
        width={width}
        height={height}
        style={{
          fill: npl_ratio !== undefined ? getColor(npl_ratio) : COLORS.primary,
          stroke: "#fff",
          strokeWidth: 2,
        }}
      />
      {width > 60 && height > 40 && (
        <>
          <text
            x={x + width / 2}
            y={y + height / 2 - 10}
            textAnchor="middle"
            fill="#fff"
            fontSize={16}
            fontWeight="bold"
          >
            {name}
          </text>
          <text
            x={x + width / 2}
            y={y + height / 2 + 10}
            textAnchor="middle"
            fill="#fff"
            fontSize={14}
          >
            {eop_balance?.toLocaleString()} tỷ
          </text>
          <text
            x={x + width / 2}
            y={y + height / 2 + 28}
            textAnchor="middle"
            fill="#fff"
            fontSize={12}
          >
            NPL:{" "}
            {npl_ratio !== undefined ? (npl_ratio * 100).toFixed(2) : "N/A"}%
          </text>
        </>
      )}
    </g>
  );
}
